#include <iostream>
#include <algorithm>
#include <climits>
using namespace std;

int n, cost[10][10];
int dp[1024][10];

int tsp(int mask, int pos) {
    // All cities are visited
    if (mask == (1 << n) - 1)
        return cost[pos][0];

    if (dp[mask][pos] != -1)
        return dp[mask][pos];

    int ans = INT_MAX;

    for (int city = 0; city < n; city++) {
        // Check whether the city is unvisited
        if ((mask & (1 << city)) == 0) {
            int newCost = cost[pos][city] +
                tsp(mask | (1 << city), city);

            ans = min(ans, newCost);
        }
    }

    return dp[mask][pos] = ans;
}

int main() {
    cout << "Enter number of cities: ";
    cin >> n;

    cout << "Enter cost matrix:\n";

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> cost[i][j];

    // Initialize DP table
    for (int i = 0; i < 1024; i++)
        for (int j = 0; j < 10; j++)
            dp[i][j] = -1;

    cout << "Minimum travelling cost: " << tsp(1, 0);

    return 0;
}
