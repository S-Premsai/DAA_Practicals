#include <iostream>
#include <vector>
#include <algorithm>

int knapsack(int W, const std::vector<int>& weights, const std::vector<int>& values, int n) {
    std::vector<int> dp(W + 1, 0);

    for (int i = 0; i < n; ++i) {
        for (int w = W; w >= weights[i]; --w) {
            dp[w] = std::max(dp[w], dp[w - weights[i]] + values[i]);
        }
    }

    return dp[W];
}

int main() {
    int n, W;
    if (!(std::cin >> n >> W)) return 0;

    std::vector<int> values(n);
    std::vector<int> weights(n);

    for (int i = 0; i < n; ++i) {
        std::cin >> values[i];
    }

    for (int i = 0; i < n; ++i) {
        std::cin >> weights[i];
    }

    std::cout << knapsack(W, weights, values, n) << std::endl;

    return 0;
}
