
#include <iostream>
#include <climits>
using namespace std;

int main() {
    int n;
    cout << "Enter number of vertices: ";
    cin >> n;

    int graph[100][100];

    cout << "Enter the adjacency matrix (0 for no edge):\n";
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> graph[i][j];
        }
    }

    int parent[100];
    int key[100];
    bool visited[100] = {false};

    for (int i = 0; i < n; i++) {
        key[i] = INT_MAX;
        parent[i] = -1;
    }

    key[0] = 0;

    for (int count = 0; count < n; count++) {
        int u = -1;
        int minKey = INT_MAX;

        for (int v = 0; v < n; v++) {
            if (!visited[v] && key[v] < minKey) {
                minKey = key[v];
                u = v;
            }
        }

        if (u == -1) {
            cout << "Graph is disconnected.\n";
            return 0;
        }

        visited[u] = true;

        for (int v = 0; v < n; v++) {
            if (graph[u][v] > 0 && !visited[v] &&
                graph[u][v] < key[v]) {
                key[v] = graph[u][v];
                parent[v] = u;
            }
        }
    }

    int totalWeight = 0;

    cout << "\nEdges in Minimum Spanning Tree:\n";
    cout << "Edge\tWeight\n";

    for (int i = 1; i < n; i++) {
        cout << parent[i] << " - " << i
             << "\t" << graph[i][parent[i]] << "\n";
        totalWeight += graph[i][parent[i]];
    }

    cout << "Total weight of MST: " << totalWeight << "\n";

    return 0;
}
